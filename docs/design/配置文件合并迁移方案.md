# 配置文件合并迁移方案

## 概述

将 `config.json` 和 `start.json` 合并为单一的 `node_config.json`，简化配置管理，提升系统稳定性。

## 迁移策略

### 第一阶段：向后兼容支持（1-2周）

#### 1.1 新增统一配置支持
- 新增 `node_config.json` 支持读取
- 保持对 `config.json` 和 `start.json` 的兼容性
- 配置优先级：`node_config.json` > `start.json` > `config.json`

#### 1.2 配置合并工具
```python
class ConfigMerger:
    """配置文件合并工具"""
    
    def merge_configs(self, node_path: str) -> dict:
        """合并 config.json 和 start.json"""
        config = self._load_config_json(node_path)
        start_config = self._load_start_json(node_path)
        
        # 合并逻辑
        merged = self._merge_configs(config, start_config)
        
        # 生成 node_config.json
        self._save_unified_config(node_path, merged)
        
        return merged
    
    def _merge_configs(self, config: dict, start_config: dict) -> dict:
        """合并两个配置文件"""
        merged = config.copy()
        
        # 处理 start.json 的多节点格式
        if "nodes" in start_config and isinstance(start_config["nodes"], list):
            for node in start_config["nodes"]:
                if node.get("name") == config.get("node_name"):
                    # 合并启动配置
                    merged.update({
                        "entry": node.get("entry", "listener.py"),
                        "python_exe": node.get("python_exe", "")
                    })
                    
                    # 合并运行时配置
                    if "config" in node:
                        merged.update(node["config"])
                    break
        
        return merged
```

#### 1.3 配置检测机制
```python
class ConfigDetector:
    """配置文件检测器"""
    
    def detect_config_type(self, node_path: str) -> str:
        """检测节点使用的配置类型"""
        config_exists = os.path.exists(os.path.join(node_path, "config.json"))
        start_exists = os.path.exists(os.path.join(node_path, "start.json"))
        unified_exists = os.path.exists(os.path.join(node_path, "node_config.json"))
        
        if unified_exists:
            return "unified"
        elif config_exists and start_exists:
            return "legacy"
        elif config_exists:
            return "config_only"
        else:
            return "none"
```

```python
class ConfigValidator:
    """配置验证工具"""
    
    def validate_unified_config(self, config: dict) -> list:
        """验证统一配置的完整性"""
        errors = []
        
        # 必需字段检查
        required_fields = ["node_name", "entry"]
        for field in required_fields:
            if field not in config:
                errors.append(f"缺少必需字段: {field}")
        
        # 参数验证
        if "parameters" in config:
            for param in config["parameters"]:
                if "name" not in param or "type" not in param:
                    errors.append("参数定义缺少 name 或 type")
        
        return errors
```

#### 2.3 迁移向导
```python
class MigrationWizard:
    """配置迁移向导"""
    
    def migrate_project(self, project_path: str):
        """迁移整个项目的配置"""
        nodes_dir = os.path.join(project_path, "nodes")
        
        for node_name in os.listdir(nodes_dir):
            node_path = os.path.join(nodes_dir, node_name)
            if os.path.isdir(node_path):
                config_type = self.config_detector.detect_config_type(node_path)
                
                if config_type == "legacy":
                    self.merger.merge_configs(node_path)
                    self.backup_legacy_configs(node_path)
    
    def backup_legacy_configs(self, node_path: str):
        """备份旧配置文件"""
        backup_dir = os.path.join(node_path, ".config_backup")
        os.makedirs(backup_dir, exist_ok=True)
        
        # 备份 config.json 和 start.json
        for filename in ["config.json", "start.json"]:
            src = os.path.join(node_path, filename)
            if os.path.exists(src):
                shutil.copy2(src, os.path.join(backup_dir, filename))
```
